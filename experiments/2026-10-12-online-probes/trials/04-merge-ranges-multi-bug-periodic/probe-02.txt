def merge_ranges(ranges):
    """Merge overlapping or adjacent inclusive integer ranges without mutation."""
    if not ranges:
        return []
    
    # Validate input ranges
    for start, end in ranges:
        if start > end:
            raise ValueError("Start exceeds end")

    ranges.sort()
